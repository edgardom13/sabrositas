eliminar un empleado

do $$
declare
  uid uuid;
begin
  select id into uid from auth.users where email = 'maryoris@sabrositas.com';

  if uid is null then
    raise notice 'Usuario no encontrado';
    return;
  end if;

  -- 1) Desvincular perfil
  delete from public.perfiles where id = uid;

  -- 2) Limpiar auth
  delete from auth.identities     where user_id = uid;
  delete from auth.sessions       where user_id = uid;
  delete from auth.refresh_tokens where user_id = uid::text;  -- ← cast a text

  -- 3) Eliminar el usuario
  delete from auth.users where id = uid;

  raise notice 'Usuario eliminado: %', uid;
end $$;
